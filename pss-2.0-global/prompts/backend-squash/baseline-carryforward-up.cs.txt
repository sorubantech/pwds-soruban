            // ---- carried forward from 20260516052920_Add_DisplayFields_In_Currency ----
            // Seeds the MasterDataTypes / MasterDatas rows the 5 Currency FK columns
            // reference. The trailing UPDATE backfills are no-ops on an empty
            // com."Currencies" but are kept verbatim so behaviour matches the old chain.
            migrationBuilder.Sql(@"
                -- 1) Seed MasterDataTypes (idempotent)
                INSERT INTO sett.""MasterDataTypes"" (""TypeName"", ""TypeCode"", ""Description"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT 'Currency Symbol Position', 'CURRENCYSYMBOLPOSITION', 'Currency symbol placement (Before/After amount)', TRUE, TRUE, FALSE, NOW()
                WHERE NOT EXISTS (SELECT 1 FROM sett.""MasterDataTypes"" WHERE ""TypeCode"" = 'CURRENCYSYMBOLPOSITION');

                INSERT INTO sett.""MasterDataTypes"" (""TypeName"", ""TypeCode"", ""Description"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT 'Currency Thousands Separator', 'CURRENCYTHOUSANDSSEPARATOR', 'Thousands separator style for currency formatting', TRUE, TRUE, FALSE, NOW()
                WHERE NOT EXISTS (SELECT 1 FROM sett.""MasterDataTypes"" WHERE ""TypeCode"" = 'CURRENCYTHOUSANDSSEPARATOR');

                INSERT INTO sett.""MasterDataTypes"" (""TypeName"", ""TypeCode"", ""Description"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT 'Currency Decimal Separator', 'CURRENCYDECIMALSEPARATOR', 'Decimal separator style for currency formatting', TRUE, TRUE, FALSE, NOW()
                WHERE NOT EXISTS (SELECT 1 FROM sett.""MasterDataTypes"" WHERE ""TypeCode"" = 'CURRENCYDECIMALSEPARATOR');

                INSERT INTO sett.""MasterDataTypes"" (""TypeName"", ""TypeCode"", ""Description"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT 'Currency Rate Source', 'CURRENCYRATESOURCE', 'Exchange rate source (Manual/Auto)', TRUE, TRUE, FALSE, NOW()
                WHERE NOT EXISTS (SELECT 1 FROM sett.""MasterDataTypes"" WHERE ""TypeCode"" = 'CURRENCYRATESOURCE');

                INSERT INTO sett.""MasterDataTypes"" (""TypeName"", ""TypeCode"", ""Description"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT 'Currency Update Frequency', 'CURRENCYUPDATEFREQUENCY', 'Auto-update frequency for currency rates', TRUE, TRUE, FALSE, NOW()
                WHERE NOT EXISTS (SELECT 1 FROM sett.""MasterDataTypes"" WHERE ""TypeCode"" = 'CURRENCYUPDATEFREQUENCY');

                -- 2) Seed MasterDatas (idempotent via NOT EXISTS check on (TypeId, DataValue))
                INSERT INTO sett.""MasterDatas"" (""MasterDataTypeId"", ""DataName"", ""DataValue"", ""DataSetting"", ""OrderBy"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT t.""MasterDataTypeId"", v.name, v.code, '', v.ord, TRUE, TRUE, FALSE, NOW()
                FROM sett.""MasterDataTypes"" t
                CROSS JOIN (VALUES ('Before', 'Before', 1), ('After', 'After', 2)) AS v(name, code, ord)
                WHERE t.""TypeCode"" = 'CURRENCYSYMBOLPOSITION'
                  AND NOT EXISTS (SELECT 1 FROM sett.""MasterDatas"" m WHERE m.""MasterDataTypeId"" = t.""MasterDataTypeId"" AND m.""DataValue"" = v.code);

                INSERT INTO sett.""MasterDatas"" (""MasterDataTypeId"", ""DataName"", ""DataValue"", ""DataSetting"", ""OrderBy"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT t.""MasterDataTypeId"", v.name, v.code, '', v.ord, TRUE, TRUE, FALSE, NOW()
                FROM sett.""MasterDataTypes"" t
                CROSS JOIN (VALUES ('Comma', 'Comma', 1), ('Period', 'Period', 2), ('Space', 'Space', 3), ('Indian', 'Indian', 4)) AS v(name, code, ord)
                WHERE t.""TypeCode"" = 'CURRENCYTHOUSANDSSEPARATOR'
                  AND NOT EXISTS (SELECT 1 FROM sett.""MasterDatas"" m WHERE m.""MasterDataTypeId"" = t.""MasterDataTypeId"" AND m.""DataValue"" = v.code);

                INSERT INTO sett.""MasterDatas"" (""MasterDataTypeId"", ""DataName"", ""DataValue"", ""DataSetting"", ""OrderBy"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT t.""MasterDataTypeId"", v.name, v.code, '', v.ord, TRUE, TRUE, FALSE, NOW()
                FROM sett.""MasterDataTypes"" t
                CROSS JOIN (VALUES ('Period', 'Period', 1), ('Comma', 'Comma', 2)) AS v(name, code, ord)
                WHERE t.""TypeCode"" = 'CURRENCYDECIMALSEPARATOR'
                  AND NOT EXISTS (SELECT 1 FROM sett.""MasterDatas"" m WHERE m.""MasterDataTypeId"" = t.""MasterDataTypeId"" AND m.""DataValue"" = v.code);

                INSERT INTO sett.""MasterDatas"" (""MasterDataTypeId"", ""DataName"", ""DataValue"", ""DataSetting"", ""OrderBy"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT t.""MasterDataTypeId"", v.name, v.code, '', v.ord, TRUE, TRUE, FALSE, NOW()
                FROM sett.""MasterDataTypes"" t
                CROSS JOIN (VALUES ('Manual', 'Manual', 1), ('Auto - API Fetch', 'Auto', 2)) AS v(name, code, ord)
                WHERE t.""TypeCode"" = 'CURRENCYRATESOURCE'
                  AND NOT EXISTS (SELECT 1 FROM sett.""MasterDatas"" m WHERE m.""MasterDataTypeId"" = t.""MasterDataTypeId"" AND m.""DataValue"" = v.code);

                INSERT INTO sett.""MasterDatas"" (""MasterDataTypeId"", ""DataName"", ""DataValue"", ""DataSetting"", ""OrderBy"", ""IsSystem"", ""IsActive"", ""IsDeleted"", ""CreatedDate"")
                SELECT t.""MasterDataTypeId"", v.name, v.code, '', v.ord, TRUE, TRUE, FALSE, NOW()
                FROM sett.""MasterDataTypes"" t
                CROSS JOIN (VALUES ('Daily', 'Daily', 1), ('Weekly', 'Weekly', 2), ('Monthly', 'Monthly', 3)) AS v(name, code, ord)
                WHERE t.""TypeCode"" = 'CURRENCYUPDATEFREQUENCY'

                -- 3) Backfill existing Currencies rows from 0 to the system-default MasterDataId.
                UPDATE com.""Currencies"" SET ""SymbolPositionId"" = (
                    SELECT m.""MasterDataId"" FROM sett.""MasterDatas"" m
                    INNER JOIN sett.""MasterDataTypes"" t ON t.""MasterDataTypeId"" = m.""MasterDataTypeId""
                    WHERE t.""TypeCode"" = 'CURRENCYSYMBOLPOSITION' AND m.""DataValue"" = 'Before'
                ) WHERE ""SymbolPositionId"" = 0;

                UPDATE com.""Currencies"" SET ""ThousandsSeparatorId"" = (
                    SELECT m.""MasterDataId"" FROM sett.""MasterDatas"" m
                    INNER JOIN sett.""MasterDataTypes"" t ON t.""MasterDataTypeId"" = m.""MasterDataTypeId""

                    INNER JOIN sett.""MasterDataTypes"" t ON t.""MasterDataTypeId"" = m.""MasterDataTypeId""
                    WHERE t.""TypeCode"" = 'CURRENCYDECIMALSEPARATOR' AND m.""DataValue"" = 'Period'
                ) WHERE ""DecimalSeparatorId"" = 0;

                    SELECT m.""MasterDataId"" FROM sett.""MasterDatas"" m
                    INNER JOIN sett.""MasterDataTypes"" t ON t.""MasterDataTypeId"" = m.""MasterDataTypeId""

            // ---- carried forward from 20260818062300_Replace_Field_Name_Unique_Index ----
            // Expression + partial unique indexes on sett."Fields". Invisible to the model
            // snapshot by design - see FieldConfiguration.cs.
            // NOT written with CREATE INDEX CONCURRENTLY: EF wraps a migration in a
            // transaction and CONCURRENTLY cannot run inside one. sett."Fields" is a small
            // configuration table.
            migrationBuilder.Sql(@"
                CREATE UNIQUE INDEX IF NOT EXISTS ux_fields_system_name
                    ON sett.""Fields"" (lower(""FieldName""))
                    WHERE ""IsSystem"" = true AND ""IsDeleted"" = false;
            ");

            // COALESCE("CompanyId", 0) rather than NULLS NOT DISTINCT (needs PostgreSQL 15)
            // folds the platform-wide NULL rows into one ordinary bucket the index can
            // police; company id 0 does not exist, so it collides with nothing real.
            migrationBuilder.Sql(@"
                CREATE UNIQUE INDEX IF NOT EXISTS ux_fields_custom_name
                    ON sett.""Fields"" (COALESCE(""CompanyId"", 0), lower(""FieldName""))
                    WHERE ""IsSystem"" = false AND ""IsDeleted"" = false;
            ");
